\section*{Abréviations principales}

\begin{description}
    \item[PFE] Projet de Fin d'Études
    \item[ENSA-M] École Nationale des Sciences Appliquées de Marrakech
    \item[ERP] Enterprise Resource Planning (Système de gestion intégré)
    \item[API] Application Programming Interface (Interface de programmation)
    \item[ORM] Object-Relational Mapping (Mapping objet-relationnel)
    \item[SGBD] Système de Gestion de Bases de Données
    \item[CLI] Command Line Interface (Interface en ligne de commande)
    \item[REST] Representational State Transfer (Architecture web)
    \item[SCRUM] Framework de gestion de projet Agile
    \item[UML] Unified Modeling Language (Langage de modélisation unifié)
    \item[CQRS] Command Query Responsibility Segregation
    \item[Delta Sync] Synchronisation incrémentale
    \item[AdminIS] ERP interne utilisé comme source de données
    \item[PowerTeam] Plateforme de gestion des fiduciaires
    \item[CI/CD] Continuous Integration / Continuous Deployment
end{description}
